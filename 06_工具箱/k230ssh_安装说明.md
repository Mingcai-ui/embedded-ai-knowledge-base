# ============================================================
# k230ssh 安装说明（备份自 WSL ~/.bashrc 追加片段）
# ============================================================
# 部署位置：WSL (Ubuntu-24.04) 用户 ~/ 下
#   1. 脚本：~/bin/k230ssh  （chmod +x）
#   2. ~/.bashrc 追加以下内容：
#
#   export PATH="$HOME/bin:$PATH"
#   alias k230ssh="~/bin/k230ssh"
#
# 用法：在 WSL 终端执行 `k230ssh`，自动确保容器 k230 在跑、
#       sshd 已启动，并打印连接命令 ssh user@<IP> -p 22222
# ============================================================

export PATH="$HOME/bin:$PATH"
alias k230ssh="~/bin/k230ssh"
